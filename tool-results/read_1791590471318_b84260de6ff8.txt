    25→type Phase = "setup" | "play" | "done";
    26→
    27→interface PickedWord extends VocabWord {

    32→
    33→/** normalize a spelled answer: lowercase, unify apostrophes, trim punctuation/spaces */
    34→function norm(s: string): string {
    35→  return s
    36→    .toLowerCase()
    37→    .replace(/[’‘`]/g, "'")
    38→    .replace(/[.,!?;:]+$/g, "")
    39→    .replace(/\s+/g, " ")
    40→    .trim();
    41→}
    42→
    43→function shuffle<T>(arr: T[]): T[] {
    44→  const a = [...arr];
    45→  for (let i = a.length - 1; i > 0; i--) {
    46→    const j = Math.floor(Math.random() * (i + 1));
    47→    [a[i], a[j]] = [a[j], a[i]];
    48→  }
    49→  return a;
    50→}

    52→export default function DictationTab({ lesson }: { lesson: LessonData }) {
    53→  const { toast } = useToast();
    54→  const mounted = useMounted();
    55→  const t = themeOf(lesson.color);
    56→
    57→  const known = useStudy((s) => s.known);
    58→  const srs = useStudy((s) => s.srs);
    59→  const reviewWord = useStudy((s) => s.reviewWord);
    60→  const userIdRef = useRef<string>("");
    61→
    62→  const [phase, setPhase] = useState<Phase>("setup");

    65→  const [words, setWords] = useState<PickedWord[]>([]);

    70→  const [replaying, setReplaying] = useState(false);
    71→  const [spelling, setSpelling] = useState(false);
    72→  const [audioErrorFor, setAudioErrorFor] = useState<number | null>(null);
    73→  const inputRef = useRef<HTMLInputElement>(null);
    74→  const advanceTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    75→
    76→  const word = words[idx];
    77→  const total = words.length;
    78→
    79→  const playAudio = useCallback(
    80→    (text: string, speed = 0.85, errTag: number | null = null) => {
    81→      setReplaying(true);
    82→      speak(
    83→        text,
    84→        speed,
    85→        () => setReplaying(false),
    86→        () => {
    87→          setReplaying(false);
    88→          setAudioErrorFor(errTag);
    89→        }
    90→      );
    91→    },
    92→    []
    93→  );
    94→
    95→  /** hear the word spelled letter-by-letter (counts as a hint → half credit) */

    99→      setWords((ws) => ws.map((w, i) => (i === idx && !w.hintUsed ? { ...w, hint: en.length, hintUsed: true } : w)));
   100→    }
   101→    setSpelling(true);
   102→    speakSpell(
   103→      en,
   104→      () => setSpelling(false),
   105→      () => setSpelling(false)
   106→    );
   107→  }, [idx]);
   108→

   110→    const base = lesson.vocabulary;
   111→    let picked = shuffle(base);
   112→    if (smartOrder && mounted) {
   113→      const now = Date.now();
   114→      picked = picked.sort((a, b) => {
   115→        const pri = (w: VocabWord) => {
   116→          const k = wordKey(lesson.id, w.en);
   117→          if (!known[k]) return 0; // unknown first
   118→          const item = srs[k];
   119→          if (item && item.dueAt <= now) return 1; // known but due
   120→          return 2; // known and scheduled later
   121→        };
   122→        return pri(a) - pri(b);
   123→      });
   124→    }
   125→    setWords(picked.slice(0, Math.min(count, base.length)).map((w) => ({ ...w, hint: 0, hintUsed: false })));
   126→    setIdx(0);
   127→    setInput("");
   128→    setState("typing");
   129→    setScore(0);
   130→    setAudioErrorFor(null);
   131→    setPhase("play");
   132→  };
   133→
   134→  // autoplay on each new word (deferred to a callback so React doesn't cascade renders)
   135→  useEffect(() => {
   136→    if (phase !== "play" || !word || state !== "typing") return;
   137→    inputRef.current?.focus();
   138→    const t = setTimeout(() => playAudio(word.en, 0.85, idx), 60);
   139→    return () => clearTimeout(t);
   140→  }, [phase, idx, state]);
   141→
   142→  // cleanup audio + pending timer on unmount / word change
   143→  useEffect(() => {
   144→    return () => {
   145→      stopSpeaking();
   146→      if (advanceTimer.current) clearTimeout(advanceTimer.current);
   147→    };
   148→  }, [idx]);
   149→
   150→  const finish = useCallback(
   151→    (finalScore: number) => {
   152→      const percent = Math.round((finalScore / total) * 100);
   153→      const correctForStats = Math.round(finalScore);
   154→      useStudy.getState().addResult(resultKey(lesson.id, "dictation"), {
   155→        correct: correctForStats,
   156→        total,
   157→        percent,
   158→        at: new Date().toISOString(),
   159→      });
   160→      useStudy.getState().trackQuizDone();
   161→      useStudy.getState().trackDictation(); // daily quest: dictation round
   162→      toast({
   163→        title: percent >= 80 ? "🏆 دیکتهٔ فوق‌العاده!" : percent >= 50 ? "👍 خوب بود، ادامه بده!" : "💪 با تمرین بهتر می‌شود",
   164→        description: `دیکتهٔ درس: ${faNum(correctForStats)} از ${faNum(total)} (${faNum(percent)}٪)`,

   168→        const store = useStudy.getState();

   173→          headers: { "Content-Type": "application/json" },
   174→          body: JSON.stringify({ userId, lesson: lesson.id, category: "dictation", total, correct: correctForStats }),
   175→        }).catch(() => {});
   176→      } catch {
   177→        /* ignore */
   178→      }
   179→      setPhase("done");
   180→    },
   181→    [lesson.id, total, toast]
   182→  );
   183→
   184→  const goNext = useCallback(
   185→    (newScore: number) => {
   186→      if (idx + 1 >= total) {
   187→        finish(newScore);
   188→      } else {
   189→        setIdx((i) => i + 1);
   190→        setInput("");
   191→        setState("typing");
   192→      }

   200→    const pts = ok ? (word.hintUsed ? 0.5 : 1) : 0;
   201→    const newScore = score + pts;
   202→    setScore(newScore);
   203→    setState(ok ? "correct" : "wrong");
   204→    setWords((ws) => ws.map((w, i) => (i === idx ? { ...w, result: ok ? (w.hintUsed ? "half" : "full") : "wrong" } : w)));
   205→    // feed the SRS engine: spelling success advances the word's review schedule
   206→    reviewWord(wordKey(lesson.id, word.en), ok && !word.hintUsed);
   207→    if (ok) {
   208→      advanceTimer.current = setTimeout(() => goNext(newScore), 1500);
   209→    }
   210→  };
   211→
   212→  const revealHint = () => {
   213→    if (!word || state !== "typing") return;
   214→    if (word.hint >= word.en.length) return;

   216→    const nextWords = [...words];
   217→    nextWords[idx] = { ...word, hint: nextHint, hintUsed: true };

   219→    setInput(word.en.slice(0, nextHint));

   227→    setState("wrong");

   231→    stopSpeaking();
   232→    setPhase("setup");
   233→    setWords([]);
   234→    setIdx(0);
   235→    setScore(0);
   236→    setSpelling(false);
   237→  };

   239→  // ----------------------------- SETUP -----------------------------

   242→      <motion.div
   243→        initial={{ opacity: 0, y: 12 }}
   244→        animate={{ opacity: 1, y: 0 }}

   246→      >
   247→        <div className="h-1.5 w-full bg-gradient-to-l from-violet-500 to-fuchsia-500" />
   248→        <div className="p-6 text-center sm:p-8">
   249→          <span className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-2xl bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-lg shadow-violet-500/30">
   250→            <Ear size={30} />
   251→          </span>
   252→          <h3 className="mb-2 text-xl font-black text-stone-800">آزمون دیکته — املای شنیداری</h3>
   253→          <p className="mx-auto mb-6 max-w-md text-[13px] leading-7 text-stone-500">
   254→            کلمه را می‌شنوید، معنی فارسی را می‌بینید و املای انگلیسی آن را تایپ می‌کنید. واژه‌های درست و غلط در
   255→            برنامهٔ مرور هوشمند (SRS) ثبت می‌شوند.
   256→          </p>
   257→
   258→          <div className="mx-auto mb-5 max-w-sm">
   259→            <div className="mb-2 text-[12px] font-bold text-stone-500">تعداد واژه‌ها</div>
   260→            <div className="flex justify-center gap-2">
   261→              {[8, 10, 15, 20].map((n) => (

   263→                  key={n}

   265→                  className={`rounded-xl px-4 py-2 text-sm font-bold transition-all active:scale-95 ${
   266→                    count === n
   267→                      ? "bg-violet-600 text-white shadow-md shadow-violet-500/25"
   268→                      : "border border-stone-200 bg-white text-stone-600 hover:border-violet-300 hover:text-violet-600"

   270→                >
   271→                  {faNum(n)}
   272→                </button>
   273→              ))}
   274→            </div>
   275→          </div>
   276→
   277→          <button
   278→            onClick={() => setSmartOrder((v) => !v)}
   279→            className={`mx-auto mb-6 flex items-center gap-2 rounded-xl border px-4 py-2 text-[12px] font-bold transition-all ${
   280→              smartOrder
   281→                ? "border-violet-300 bg-violet-50 text-violet-700"
   282→                : "border-stone-200 bg-white text-stone-500 hover:text-violet-600"
   283→            }`}
   284→          >
   285→            <Sparkles size={14} />
   286→            اولویت با واژه‌های یادنگرفته و سررسید
   287→            <span className={`h-4 w-7 rounded-full p-0.5 transition-colors ${smartOrder ? "bg-violet-600" : "bg-stone-300"}`}>
   288→              <span className={`block h-3 w-3 rounded-full bg-white transition-transform ${smartOrder ? "-translate-x-3" : ""}`} />
   289→            </span>
   290→          </button>
   291→
   292→          <button
   293→            onClick={startRound}
   294→            className="inline-flex items-center gap-2 rounded-xl bg-gradient-to-l from-violet-600 to-fuchsia-600 px-7 py-3 text-sm font-bold text-white shadow-lg shadow-violet-600/30 transition-all hover:opacity-90 active:scale-95"
   295→          >
   296→            <Play size={16} />
   297→            شروع دیکته
   298→          </button>
   299→        </div>
   300→      </motion.div>
   301→    );
   302→  }
   303→
   304→  // ----------------------------- DONE -----------------------------
   305→  if (phase === "done") {
   306→    const percent = Math.round((score / total) * 100);
   307→    const noHint = words.filter((w) => w.result === "full").length;
   308→    const half = words.filter((w) => w.result === "half").length;
   309→    const wrong = words.filter((w) => w.result === "wrong").length;
   310→    return (
   311→      <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="space-y-4">
   312→        <div className="overflow-hidden rounded-2xl border border-stone-200 bg-white p-8 text-center shadow-sm">
   313→          <span className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-2xl bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-lg shadow-violet-500/30">
   314→            <Trophy size={28} />
   315→          </span>
   316→          <h3 className="mb-2 text-xl font-black text-stone-800">دیکته تمام شد!</h3>
   317→          <p className="mb-1 text-4xl font-black text-violet-600">{faNum(percent)}٪</p>
   318→          <p className="mb-2 text-sm text-stone-500">
   319→            امتیاز {faNum(score)} از {faNum(total)}
   320→          </p>
   321→          <div className="mb-6 flex flex-wrap justify-center gap-2 text-[12px] font-bold">
   322→            <span className="rounded-full bg-emerald-50 px-3 py-1 text-emerald-700">کامل: {faNum(noHint)}</span>
   323→            <span className="rounded-full bg-amber-50 px-3 py-1 text-amber-700">با راهنما: {faNum(half)}</span>
   324→            <span className="rounded-full bg-rose-50 px-3 py-1 text-rose-700">غلط: {faNum(wrong)}</span>
   325→          </div>
   326→          <div className="flex flex-wrap justify-center gap-3">
   327→            <button
   328→              onClick={startRound}
   329→              className="inline-flex items-center gap-2 rounded-xl bg-violet-600 px-5 py-2.5 text-sm font-bold text-white shadow-md transition-all hover:bg-violet-700 active:scale-95"
   330→            >
   331→              <RotateCcw size={15} />

   333→            </button>

   337→            >
   338→              بازگشت به تنظیمات
   339→            </button>
   340→          </div>
   341→        </div>

   343→        {/* review of all asked words */}
   344→        <div className="rounded-2xl border border-stone-200 bg-white p-4 shadow-sm">
   345→          <h4 className="mb-3 text-sm font-extrabold text-stone-700">واژه‌های این دور</h4>
   346→          <div className="grid max-h-72 gap-2 overflow-y-auto nice-scroll sm:grid-cols-2">
   347→            {words.map((w) => (

   349→                key={w.en}
   350→                className={`flex items-center justify-between rounded-xl border p-3 ${
   351→                  w.result === "full"
   352→                    ? "border-emerald-200 bg-emerald-50/50"
   353→                    : w.result === "half"

   355→                      : "border-rose-200 bg-rose-50/50"
   356→                }`}

   360→                    <CheckCircle2 size={16} className="shrink-0 text-emerald-600" />

   364→                    <XCircle size={16} className="shrink-0 text-rose-500" />
   365→                  )}
   366→                  <div className="min-w-0">
   367→                    <div className="en-text truncate text-sm font-bold text-stone-800" dir="ltr">
   368→                      {w.en}
   369→                    </div>
   370→                    <div className="truncate text-[12px] text-stone-500">{w.fa}</div>
   371→                  </div>
   372→                </div>
   373→                <div className="mr-2 flex shrink-0 items-center gap-1.5">

   377→                    aria-label={`روخوانی ${w.en}`}

   384→                    aria-label={`املای حرف‌به‌حرف ${w.en}`}

   387→                    <SpellCheck size={14} />
   388→                  </button>

   392→          </div>
   393→        </div>
   394→      </motion.div>
   395→    );
   396→  }
   397→
   398→  // ----------------------------- PLAY -----------------------------
   399→  if (!word) return null;
   400→
   401→  const answerChars = word.en.split("");
   402→  const userChars = norm(input).split("");
   403→  const inputChars = input.split("");
   404→  const isCorrect = state === "correct";
   405→
   406→  return (
   407→    <div className="space-y-4">
   408→      {/* progress */}
   409→      <div className="flex items-center gap-3">
   410→        <div className="h-2 flex-1 overflow-hidden rounded-full bg-stone-200">
   411→          <div
   412→            className="h-full bg-gradient-to-l from-violet-500 to-fuchsia-500 transition-all duration-300"
   413→            style={{ width: `${((idx + (state === "typing" ? 0 : 1)) / total) * 100}%` }}
   414→          />
   415→        </div>
   416→        <span className="rounded-lg bg-violet-50 px-2.5 py-1 text-[12px] font-bold text-violet-700">
   417→          {faNum(Math.min(idx + 1, total))} از {faNum(total)}
   418→        </span>
   419→        <span className="rounded-lg bg-stone-100 px-2.5 py-1 text-[12px] font-bold text-stone-600">
   420→          امتیاز: {faNum(score)}
   421→        </span>
   422→      </div>
   423→
   424→      <AnimatePresence mode="wait">
   425→        <motion.div
   426→          key={idx}
   427→          initial={{ opacity: 0, x: -18 }}
   428→          animate={{ opacity: 1, x: 0 }}

   430→          transition={{ duration: 0.25 }}

   434→              : isCorrect

   436→                : "border-rose-300"

   446→              className="relative flex h-20 w-20 items-center justify-center rounded-full bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white shadow-xl shadow-violet-500/35 transition-all hover:opacity-90"
