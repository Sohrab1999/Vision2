    14→    14→export interface QuizEngineProps {

    17→    17→  resultKeyStr: string; // e.g. "1:vocab" or "exam"

    35→    35→}
    36→    36→
    37→    37→const letters = ["a", "b", "c", "d"];
    38→    38→
    39→    39→export default function QuizEngine({
    40→    40→  title,
    41→    41→  questions,
    42→    42→  resultKeyStr,
    43→    43→  lessonId,
    44→    44→  category,
    45→    45→  immediate = true,
    46→    46→  secondsPerQuestion,
    47→    47→  onExit,
    48→    48→  groupOf,
    49→    49→  groupLabels,
    50→    50→  certificate = false,
    51→    51→  mistakesOf,
    52→    52→  trackMistakes = true,
    53→    53→  faOptions,
    54→    54→}: QuizEngineProps) {
    55→    55→  const [idx, setIdx] = useState(0);
    56→    56→  const [selected, setSelected] = useState<number | null>(null);
    57→    57→  const [answers, setAnswers] = useState<number[]>([]); // chosen per question (in order)
    58→    58→  const [finished, setFinished] = useState(false);
    59→    59→  const finishedRef = useRef(false);
    60→    60→  const [timeLeft, setTimeLeft] = useState<number | null>(
    61→    61→    secondsPerQuestion ? secondsPerQuestion * questions.length : null
    62→    62→  );
    63→    63→  const [imgBusy, setImgBusy] = useState(false);
    64→    64→  const userIdRef = useRef<string>("");
    65→    65→  const { toast } = useToast();
    66→    66→
    67→    67→  const correctCount = useMemo(
    68→    68→    () => answers.reduce((s, a, i) => (a === questions[i]?.answer ? s + 1 : s), 0),
    69→    69→    [answers, questions]
    70→    70→  );
    71→    71→
    72→    72→  const q = questions[idx];
    73→    73→  const total = questions.length;
    74→    74→  const faOpt = q ? (faOptions?.(q) ?? false) : false;
    75→    75→
    76→    76→  const finish = (finalAnswers: number[]) => {
    77→    77→    if (finishedRef.current) return;
    78→    78→    finishedRef.current = true;

    80→    80→      (s, a, i) => (a === questions[i]?.answer ? s + 1 : s),

    84→    84→    useStudy.getState().addResult(resultKeyStr, {
    85→    85→      correct,
    86→    86→      total,
    87→    87→      percent,
    88→    88→      at: new Date().toISOString(),
    89→    89→    });

    91→    91→    // error bank: record wrong answers, clear ones now answered correctly
    92→    92→    const wrongIds = questions
    93→    93→      .filter((qq, i) => finalAnswers[i] !== qq.answer)
    94→    94→      .map((qq) => qq.id);
    95→    95→    const rightIds = questions
    96→    96→      .filter((qq, i) => finalAnswers[i] === qq.answer)
    97→    97→      .map((qq) => qq.id);
    98→    98→    const storeNow = useStudy.getState();

   100→   100→      if (mistakesOf) {
   101→   101→        // bucket wrong/right ids per source lesson, then record each bucket

   103→   103→        const rightBy = new Map<string, string[]>();
   104→   104→        for (const [i, qq] of questions.entries()) {
   105→   105→          const target = String(mistakesOf(qq));
   106→   106→          const bucket = finalAnswers[i] !== qq.answer ? wrongBy : rightBy;
   107→   107→          bucket.set(target, [...(bucket.get(target) ?? []), qq.id]);
   108→   108→        }
   109→   109→        for (const [target, ids] of wrongBy) storeNow.addMistakes(target, ids);
   110→   110→        for (const [target, ids] of rightBy) for (const id of ids) storeNow.clearMistake(target, id);
   111→   111→      } else {
   112→   112→        if (wrongIds.length) storeNow.addMistakes(lessonId, wrongIds);
   113→   113→        for (const id of rightIds) storeNow.clearMistake(lessonId, id);
   114→   114→      }
   115→   115→    }
   116→   116→    // F7 — word-level risk tracking from every answered vocab question
   117→   117→    try {
   118→   118→      recordQuizWordResults(questions, finalAnswers);
   119→   119→    } catch {
   120→   120→      /* analysis is best-effort — never block the result */
   121→   121→    }
   122→   122→    toast({
   123→   123→      title: percent >= 70 ? "🏆 نتیجهٔ عالی!" : percent >= 50 ? "👍 نتیجهٔ قابل قبول" : "💪 تمرین بیشتر!",
   124→   124→      description: `${title}: ${correct} از ${total} درست (${percent}٪)${
   125→   125→        percent >= 90 ? " — فوق‌العاده بود!" : ""
   126→   126→      }`,
   127→   127→      duration: 4000,
   128→   128→    });
   129→   129→    // fire-and-forget to backend
   130→   130→    try {
   131→   131→      const store = useStudy.getState();
   132→   132→      const userId = userIdRef.current || store.ensureUserId();
   133→   133→      userIdRef.current = userId;
   134→   134→      fetch("/api/stats", {
   135→   135→        method: "POST",
   136→   136→        headers: { "Content-Type": "application/json" },
   137→   137→        body: JSON.stringify({
   138→   138→          userId,
   139→   139→          lesson: typeof lessonId === "number" ? lessonId : 0,
   140→   140→          category,
   141→   141→          total,
   142→   142→          correct,
   143→   143→        }),
   144→   144→      }).catch(() => {});
   145→   145→    } catch {
   146→   146→      /* ignore */
   147→   147→    }
   148→   148→    // freeze the answers so late clicks during exit animations can't mutate the result
   149→   149→    setAnswers(finalAnswers);
   150→   150→    setFinished(true);
   151→   151→  };
   152→   152→
   153→   153→  // timer
   154→   154→  useEffect(() => {
   155→   155→    if (timeLeft === null || finished) return;

   157→   157→      const next = timeLeft - 1;
   158→   158→      setTimeLeft(next);
   159→   159→      if (next <= 0) {
   160→   160→        finish(answers);
   161→   161→      }
   162→   162→    }, 1000);
   163→   163→    return () => clearTimeout(t);
   164→   164→  }, [timeLeft, finished]);
   165→   165→
   166→   166→  const choose = (i: number) => {
   167→   167→    if (finishedRef.current || finished) return;
   168→   168→    if (immediate) {
   169→   169→      if (selected !== null) return;
   170→   170→      const next = [...answers];
   171→   171→      next[idx] = i;
   172→   172→      setAnswers(next);
   173→   173→      setSelected(i);
   174→   174→    } else {
   175→   175→      // exam mode: record answer and auto advance (one answer per question)
   176→   176→      if (answers[idx] !== undefined) return;
   177→   177→      const next = [...answers];
   178→   178→      next[idx] = i;
   179→   179→      setAnswers(next);
   180→   180→      if (idx < total - 1) {
   181→   181→        setIdx(idx + 1);
   182→   182→        setSelected(null);
   183→   183→      } else {
   184→   184→        finish(next);
   185→   185→      }
   186→   186→    }
   187→   187→  };
   188→   188→
   189→   189→  const nextQuestion = () => {
   190→   190→    if (idx < total - 1) {
   191→   191→      setIdx(idx + 1);
   192→   192→      setSelected(null);
   193→   193→    } else {
   194→   194→      finish(answers);
   195→   195→    }
   196→   196→  };
   197→   197→
   198→   198→  const restart = () => {
   199→   199→    finishedRef.current = false;
   200→   200→    setIdx(0);
   201→   201→    setSelected(null);
   202→   202→    setAnswers([]);
   203→   203→    setFinished(false);
   204→   204→    setTimeLeft(secondsPerQuestion ? secondsPerQuestion * questions.length : null);
   205→   205→  };
   206→   206→
   207→   207→  // keyboard shortcuts: 1-4 choose an option, Enter / ← advance
   208→   208→  // (re-registered each render so closures are always fresh)
   209→   209→  useEffect(() => {
   210→   210→    if (finished) return;
   211→   211→    const onKey = (e: KeyboardEvent) => {
   212→   212→      const tag = (e.target as HTMLElement | null)?.tagName;
   213→   213→      if (tag === "INPUT" || tag === "TEXTAREA") return;
   214→   214→      const num = Number(e.key);
   215→   215→      if (num >= 1 && num <= (questions[idx]?.options.length ?? 0)) {
   216→   216→        choose(num - 1);
   217→   217→        return;
   218→   218→      }
   219→   219→      if (e.key === "Enter" || e.key === "ArrowLeft") {
   220→   220→        if (immediate && selected !== null) nextQuestion();
   221→   221→        else if (!immediate) {
   222→   222→          if (idx < total - 1) {
   223→   223→            setIdx(idx + 1);
   224→   224→            setSelected(null);
   225→   225→          } else {
   226→   226→            finish(answers);
   227→   227→          }
   228→   228→        }
   229→   229→      }
   230→   230→    };
   231→   231→    window.addEventListener("keydown", onKey);
   232→   232→    return () => window.removeEventListener("keydown", onKey);
   233→   233→  });
   234→   234→
   235→   235→  // scroll the feedback box into view when it appears (matters inside dialogs)
   236→   236→  useEffect(() => {
   237→   237→    if (selected === null || finished) return;
   238→   238→    document.querySelector("[data-quiz-feedback]")?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   239→   239→  }, [selected, finished]);
   240→   240→
   241→   241→  /* ------------------------------- finished ------------------------------- */
   242→   242→  if (finished) {
   243→   243→    const percent = Math.round((correctCount / total) * 100);
   244→   244→    const wrongIds = questions.filter((qq, i) => answers[i] !== qq.answer).map((qq) => qq.id);
   245→   245→    const grade =
   246→   246→      percent >= 90
   247→   247→        ? { msg: "فوق‌العاده! تو یک قهرمان زبان انگلیسی‌ای! 🏆", cls: "text-emerald-600" }
   248→   248→        : percent >= 70
   249→   249→          ? { msg: "خیلی خوب! با کمی تمرین بیشتر به عالی می‌رسی.", cls: "text-emerald-600" }
   250→   250→          : percent >= 50
   251→   251→            ? { msg: "بد نیست! نکات گرامری و واژگان را دوباره مرور کن.", cls: "text-amber-600" }
   252→   252→            : { msg: "نیاز به تمرین بیشتر داری. فلش‌کارت‌ها را مرور کن!", cls: "text-rose-600" };
   253→   253→
   254→   254→    // per-group accuracy (e.g. per source lesson in the final exam)
   255→   255→    const groups =
   256→   256→      groupOf && groupLabels
   257→   257→        ? Object.keys(groupLabels)
   258→   258→            .map((key) => {
   259→   259→              const qs = questions.filter((qq) => groupOf(qq) === key);
   260→   260→              const ok = qs.filter((qq, i) => answers[questions.indexOf(qq)] === qq.answer).length;
   261→   261→              return { key, label: groupLabels[key], total: qs.length, correct: ok };
   262→   262→            })
   263→   263→            .filter((g) => g.total > 0)
   264→   264→        : [];
   265→   265→
   266→   266→    const saveResultImage = async (premium: boolean) => {
   267→   267→      if (imgBusy) return;
   268→   268→      setImgBusy(true);
   269→   269→      const ok = await downloadResultImage({
   270→   270→        title,
   271→   271→        percent,
   272→   272→        correct: correctCount,
   273→   273→        total,
   274→   274→        userName: useStudy.getState().userName || undefined,
   275→   275→        premium,
   276→   276→      });
   277→   277→      setImgBusy(false);
   278→   278→      toast(
   279→   279→        ok
   280→   280→          ? { title: "🖼 تصویر نتیجه ذخیره شد", description: "حالا می‌توانی آن را در هر کجا به اشتراک بگذاری.", duration: 3000 }
   281→   281→          : { title: "خطا در ساخت تصویر", description: "لطفاً دوباره تلاش کن.", duration: 3000 }
   282→   282→      );
   283→   283→    };
   284→   284→
   285→   285→    const shareResult = async () => {
   286→   286→      const breakdown = groups.map((g) => `${g.label}: ${faNum(Math.round((g.correct / g.total) * 100))}٪`).join(" · ");
   287→   287→      const text = `🏆 نتیجهٔ من در وب‌کتاب «Vision Two»\n${title}\n⭐ ${faNum(percent)}٪ (${faNum(correctCount)} از ${faNum(total)})${breakdown ? `\n📊 ${breakdown}` : ""}`;
   288→   288→      try {
   289→   289→        if (typeof navigator !== "undefined" && typeof navigator.share === "function") {
   290→   290→          await navigator.share({ title: "نتیجهٔ وب‌کتاب Vision Two", text });
   291→   291→          return;
   292→   292→        }
   293→   293→        await navigator.clipboard.writeText(text);
   294→   294→        toast({ title: "✓ نتیجه کپی شد", description: "می‌توانی آن را برای دوستانت بفرستی!", duration: 3000 });
   295→   295→      } catch {
   296→   296→        /* user cancelled share */
   297→   297→      }
   298→   298→    };
   299→   299→
   300→   300→    const certificateCard = (
   301→   301→      <div className="certificate-card rounded-2xl bg-white p-1.5 shadow-xl">
   302→   302→        <div className="rounded-xl border-2 border-dashed border-amber-300 p-5 text-center sm:p-6">
   303→   303→          <div className="mb-3 flex items-center justify-center gap-2 text-[11px] font-black tracking-widest text-amber-600">
   304→   304→            <ScrollText size={14} />
   305→   305→            WEBBOOK CERTIFICATE · وب‌کتاب ویژن دو
   306→   306→          </div>
   307→   307→          <div className="mb-4 flex justify-center">
   308→   308→            <ProgressRing percent={percent} size={116} stroke={10}>
   309→   309→              <div>
   310→   310→                <div className="text-2xl font-black text-stone-800">{faNum(percent)}٪</div>
   311→   311→                <div className="text-[11px] text-stone-500" dir="ltr">
   312→   312→                  {correctCount}/{total}
   313→   313→                </div>
   314→   314→              </div>
   315→   315→            </ProgressRing>
   316→   316→          </div>
   317→   317→          <div className="mx-auto mb-3 flex w-fit items-center gap-2 rounded-full bg-gradient-to-l from-amber-400 to-orange-500 px-4 py-1.5 text-[13px] font-black text-white shadow-lg shadow-amber-500/30">
   318→   318→            <Award size={16} />
   319→   319→            {title}
   320→   320→          </div>
   321→   321→          <p className={`mb-1 text-sm font-bold ${grade.cls}`}>{grade.msg}</p>
   322→   322→          <p className="mb-4 text-[11.5px] text-stone-400">
   323→   323→            تاریخ اخذ: {new Date().toLocaleDateString("fa-IR")} · وب‌کتاب تعاملی زبان انگلیسی یازدهم
   324→   324→          </p>
   325→   325→
   326→   326→          {/* per-source breakdown */}
   327→   327→          {groups.length > 0 && (
   328→   328→            <div className="mx-auto mb-4 grid max-w-md gap-2 text-right" dir="rtl">
   329→   329→              {groups.map((g) => {
   330→   330→                const pct = Math.round((g.correct / g.total) * 100);
   331→   331→                const bar = pct >= 70 ? "from-emerald-400 to-emerald-600" : pct >= 50 ? "from-amber-400 to-orange-500" : "from-rose-400 to-rose-600";
   332→   332→                return (
   333→   333→                  <div key={g.key} className="rounded-xl border border-stone-100 bg-stone-50 px-3 py-2">
   334→   334→                    <div className="mb-1 flex items-center justify-between text-[11.5px] font-bold">
   335→   335→                      <span className="text-stone-600">{g.label}</span>
   336→   336→                      <span className="text-stone-800">
   337→   337→                        {faNum(g.correct)}/{faNum(g.total)} — {faNum(pct)}٪
   338→   338→                      </span>
   339→   339→                    </div>
   340→   340→                    <div className="h-1.5 overflow-hidden rounded-full bg-stone-200">
   341→   341→                      <motion.div
   342→   342→                        initial={{ width: 0 }}
   343→   343→                        animate={{ width: `${pct}%` }}
   344→   344→                        transition={{ duration: 0.7, ease: "easeOut" }}
   345→   345→                        className={`h-full rounded-full bg-gradient-to-l ${bar}`}
   346→   346→                      />
   347→   347→                    </div>
   348→   348→                  </div>
   349→   349→                );
   350→   350→              })}
   351→   351→            </div>
   352→   352→          )}
   353→   353→
   354→   354→          <div className="flex flex-wrap justify-center gap-2.5">
   355→   355→            <button
   356→   356→              onClick={shareResult}
   357→   357→              className="inline-flex items-center gap-1.5 rounded-xl bg-gradient-to-l from-amber-500 to-orange-600 px-4 py-2.5 text-sm font-bold text-white shadow-md shadow-amber-600/25 transition-all hover:opacity-90 active:scale-95"
   358→   358→            >
   359→   359→              <Share2 size={15} />
   360→   360→              اشتراک‌گذاری نتیجه
   361→   361→            </button>
   362→   362→            <button

   364→   364→              disabled={imgBusy}

   375→   375→              تلاش مجدد
   376→   376→            </button>

   378→   378→              onClick={onExit}
   379→   379→              className="inline-flex items-center gap-1.5 rounded-xl border border-stone-300 bg-white px-4 py-2.5 text-sm font-bold text-stone-600 transition-all hover:bg-stone-50 active:scale-95"
   380→   380→            >
   381→   381→              بازگشت
   382→   382→            </button>
   383→   383→          </div>
   384→   384→        </div>
   385→   385→      </div>
   386→   386→    );
   387→   387→
   388→   388→    const standardCard = (
   389→   389→      <div className="rounded-2xl border border-stone-200 bg-white p-6 text-center shadow-sm">

   393→   393→              <div className="text-2xl font-black text-stone-800">{percent}٪</div>
   394→   394→              <div className="text-[11px] text-stone-500" dir="ltr">
   395→   395→                {correctCount}/{total}
   396→   396→              </div>
   397→   397→            </div>
   398→   398→          </ProgressRing>
   399→   399→        </div>
   400→   400→        <div className="mb-1 flex items-center justify-center gap-2 text-lg font-black">
