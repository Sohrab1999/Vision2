    33→    33→/** normalize a spelled answer: lowercase, unify apostrophes, trim punctuation/spaces */
    34→    34→function norm(s: string): string {
    35→    35→  return s
    36→    36→    .toLowerCase()
    37→    37→    .replace(/[’‘`]/g, "'")
    38→    38→    .replace(/[.,!?;:]+$/g, "")
    39→    39→    .replace(/\s+/g, " ")
    40→    40→    .trim();
    41→    41→}
    42→    42→
    43→    43→function shuffle<T>(arr: T[]): T[] {
    44→    44→  const a = [...arr];
    45→    45→  for (let i = a.length - 1; i > 0; i--) {
    46→    46→    const j = Math.floor(Math.random() * (i + 1));
    47→    47→    [a[i], a[j]] = [a[j], a[i]];
    48→    48→  }
    49→    49→  return a;
    50→    50→}
